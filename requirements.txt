@echo off
title Data Quality Project - Install Dependencies

echo Creating Python virtual environment...
python -m venv .venv
if errorlevel 1 (
    echo Python was not found. Try installing Python or use "py" instead of "python".
    pause
    exit /b 1
)

echo Activating virtual environment...
call .venv\Scripts\activate

echo Upgrading pip...
python -m pip install --upgrade pip

echo Installing packages from requirements.txt...
python -m pip install -r requirements.txt
if errorlevel 1 (
    echo Installation failed. Check the error above.
    pause
    exit /b 1
)

echo.
echo All dependencies were installed successfully.
echo You can now open the notebooks in the notebooks folder.
pause
